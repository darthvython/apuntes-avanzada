mkdir -p ~/workspace
cat << 'SCRIPT_EOF' > ~/workspace/setup_keys.sh
#!/bin/bash

# 1. Aplicar teclado English (US, intl., with dead keys)
setxkbmap -layout us -variant intl

# 2. Detectar el archivo real de configuración de Openbox
mkdir -p ~/.config/openbox

if [ -n "$XDG_CURRENT_DESKTOP" ] && echo "$XDG_CURRENT_DESKTOP" | grep -qi "lxqt"; then
    OB_FILE="$HOME/.config/openbox/lxqt-rc.xml"
    TEMPLATE="/etc/xdg/openbox/lxqt-rc.xml"
else
    OB_FILE="$HOME/.config/openbox/lubuntu-rc.xml"
    TEMPLATE="/etc/xdg/openbox/lubuntu-rc.xml"
fi

# Fallback si no existe la plantilla específica
if [ ! -f "$TEMPLATE" ]; then
    TEMPLATE="/etc/xdg/openbox/rc.xml"
    [ ! -f "$OB_FILE" ] && OB_FILE="$HOME/.config/openbox/rc.xml"
fi

# Copiar plantilla si el usuario no tiene config propia
if [ ! -f "$OB_FILE" ] && [ -f "$TEMPLATE" ]; then
    cp "$TEMPLATE" "$OB_FILE"
fi

if [ ! -f "$OB_FILE" ]; then
    echo "ERROR: no se encontró ningún rc.xml de Openbox" >&2
    exit 1
fi

# 3. Limpiar keybinds previos de W-Left y W-Right (cualquier atributo extra)
sed -i '/<keybind key="W-Left"/,/<\/keybind>/d' "$OB_FILE"
sed -i '/<keybind key="W-Right"/,/<\/keybind>/d' "$OB_FILE"

# 4. Insertar las acciones nativas
cat << 'XML_EOF' > /tmp/openbox_keys.xml
  <keybind key="W-Left">
    <action name="Unmaximize"/>
    <action name="MoveResizeTo">
      <x>0</x>
      <y>0</y>
      <width>50%</width>
      <height>100%</height>
    </action>
  </keybind>
  <keybind key="W-Right">
    <action name="Unmaximize"/>
    <action name="MoveResizeTo">
      <x>-0</x>
      <y>0</y>
      <width>50%</width>
      <height>100%</height>
    </action>
  </keybind>
XML_EOF

sed -i '/<\/keyboard>/r /tmp/openbox_keys.xml' "$OB_FILE"
rm -f /tmp/openbox_keys.xml

# 5. Recargar Openbox
openbox --reconfigure
SCRIPT_EOF
