#!/bin/bash

# 1. Teclado English (US, intl., with dead keys) - La opción de tu foto
setxkbmap -layout us -variant intl

# 2. Ruta de configuración de Openbox
mkdir -p ~/.config/openbox
CONFIG_FILE="$HOME/.config/openbox/lxqt-rc.xml"

# Si no existe localmente, copiar la plantilla base del sistema
if [ ! -f "$CONFIG_FILE" ]; then
    if [ -f "/etc/xdg/openbox/lxqt-rc.xml" ]; then
        cp /etc/xdg/openbox/lxqt-rc.xml "$CONFIG_FILE"
    elif [ -f "/etc/xdg/openbox/rc.xml" ]; then
        cp /etc/xdg/openbox/rc.xml "$CONFIG_FILE"
    fi
fi

# 3. Limpiar atajos previos si existían
sed -i '/<keybind key="W-Left">/,/<\/keybind>/d' "$CONFIG_FILE"
sed -i '/<keybind key="W-Right">/,/<\/keybind>/d' "$CONFIG_FILE"

# 4. Insertar los atajos Win + Flecha Izquierda / Derecha
cat << 'EOF' > /tmp/openbox_keys.xml
  <keybind key="W-Left">
    <action name="UnmaximizeFull"/>
    <action name="MaximizeVert"/>
    <action name="MoveResizeTo">
      <x>0</x>
      <y>0</y>
      <width>50%</width>
      <height>100%</height>
    </action>
  </keybind>
  <keybind key="W-Right">
    <action name="UnmaximizeFull"/>
    <action name="MaximizeVert"/>
    <action name="MoveResizeTo">
      <x>-0</x>
      <y>0</y>
      <width>50%</width>
      <height>100%</height>
    </action>
  </keybind>
EOF

sed -i '/<\/keyboard>/r /tmp/openbox_keys.xml' "$CONFIG_FILE"
rm /tmp/openbox_keys.xml

# 5. Aplicar cambios inmediatamente
openbox --reconfigure

echo "Configuración aplicada: Teclado US Intl (dead keys) y atajos Win+Flechas."
