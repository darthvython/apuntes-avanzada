#!/bin/bash

# 1. Aplicar teclado English (US, intl., with dead keys)
setxkbmap -layout us -variant intl

# 2. Configurar atajos directamente en LXQt
SHORTCUT_DIR="$HOME/.config/lxqt"
mkdir -p "$SHORTCUT_DIR"

# Escribir la configuración de atajos en el archivo de LXQt
cat << 'EOF' > "$SHORTCUT_DIR/shortcut-keys.conf"
[Control%2BSuper%2BLeft.1]
comment=Tile window to left
exec=openbox --send-action "UnmaximizeFull MaximizeVert MoveResizeTo x=0 y=0 width=50% height=100%"

[Super%2BLeft.2]
comment=Tile window to left (Win + Left)
exec=openbox --send-action "UnmaximizeFull MaximizeVert MoveResizeTo x=0 y=0 width=50% height=100%"

[Control%2BSuper%2BRight.3]
comment=Tile window to right
exec=openbox --send-action "UnmaximizeFull MaximizeVert MoveResizeTo x=0 y=0 width=50% height=100%"

[Super%2BRight.4]
comment=Tile window to right (Win + Right)
exec=openbox --send-action "UnmaximizeFull MaximizeVert MoveResizeTo x=-0 y=0 width=50% height=100%"
EOF

# 3. Reiniciar el demonio de atajos de LXQt para que tome el archivo sin reiniciar sesión
killall lxqt-globalkeysd 2>/dev/null
lxqt-globalkeysd &

echo "Configuración aplicada."
