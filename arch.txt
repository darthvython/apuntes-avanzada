#!/bin/bash

# 1. Configurar teclado (La variante de tu foto)
setxkbmap -layout us -variant intl

# 2. Configurar atajos usando los comandos de xdotool directamente en LXQt
# Esto crea/sobreescribe los atajos de usuario
mkdir -p "$HOME/.config/lxqt"

# Definimos los comandos en el archivo de configuración de atajos de LXQt
cat << 'EOF' > "$HOME/.config/lxqt/globalkeyshortcuts.conf"
[General]
AllowGrab=true

[left_snap]
Comment=Snap Left
Shortcut=Super+Left
Command=xdotool getactivewindow windowsize 50% 100% windowmove 0 0

[right_snap]
Comment=Snap Right
Shortcut=Super+Right
Command=xdotool getactivewindow windowsize 50% 100% windowmove 50% 0
EOF

# 3. Reiniciar el gestor de atajos para aplicar
killall lxqt-globalkeysd 2>/dev/null
lxqt-globalkeysd &

echo "Configuración aplicada. Prueba Win+Flechas."
